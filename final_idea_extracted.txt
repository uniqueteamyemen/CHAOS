5. مواصفات تصميم محول الإشارات الهيكلية CausalSignalAdapter

5.1 بنية واجهة الربط البرمجي (Decoupled TypeScript Interface)
يتحكم الـ CausalSignalAdapter في استقبال البيانات مجهولة المصدر والتصنيف، ويفك ارتباط المحرك التحليلي عن أي مسميات صلبة خاصة بالمنتجات أو المواقع، معتمداً على واجهة برمجية مرنة:

// src/connectors/CausalSignalAdapter.ts

import { DirectObservationAdapter } from "market-intelligence-engine";

export interface StructuralPayload {

  patternId: string;

  triggerEventClass: string;

  sourceContext: string;

  

  // تصحيح هيكلي: توصيف عام للهدف يفصل المحرك عن مسميات العناصر أو المواقع المحددة مسبقاً

  targetEntity: {

    domainSpace: string;        // مثال: "geography_inventory", "financial_index", "asset_class"

    primaryIdentifier: string;  // مثال: "in-zone-4", "ticker-xyz", "commodity-faci"

    secondaryIdentifier?: string;

  };

  

  calculatedTimeLagMonths: number;

  

  // تصحيح هيكلي: فصل المتانة الإحصائية والقدرة التنبؤية عن ادعاءات السببية المطلقة

  statisticalRobustnessScore: number;  // درجة ثبات النمط خارج العينة التاريخية (0.00 - 1.00)

  permutationSurvivalRate: number;     // نسبة نجاح صمود النمط ضد اختبار التعمية وخلط التواريخ

  

  estimatedImpactMagnitude: number;

  

  // تصحيح هيكلي: تصنيف ديناميكي مرن ونوعي للإشارة بدلاً من فرض نوع ثابت

  signalClassification: "supply" | "demand" | "pressure_modifier" | "systemic_shock";

  timestampDiscovered: string;

}

export class CausalSignalAdapter extends DirectObservationAdapter {

  public readonly id = "structural_autonomous_inference_adapter";

  /**

   * يحول الأنماط الإحصائية المجردة المكتشفة خارجياً إلى لقطات وإشارات إحصائية محوكمة

   * داخل مخزن البيانات المستقر للمحرك الأب.

   */

  public async ingestAutonomousPattern(

    sourceId: string, 

    payload: StructuralPayload, 

    runtimeContext: any

  ): Promise<void> {

    

    // 1. التحقق الصارم من متانة الإشارة بناءً على فلاتر التحصين ضد الصدفة

    if (payload.permutationSurvivalRate < 0.99 || payload.statisticalRobustnessScore < 0.99) {

      throw new Error(`Admission Denied: Signal payload failed to survive the adversarial shuffle verification threshold.`);

    }

    const compiledSummary = `[STRUCTURAL PATTERN ISOLATED] Precedence relationship verified from trigger [${payload.triggerEventClass}] to target space [${payload.targetEntity.domainSpace}::${payload.targetEntity.primaryIdentifier}]. Transmission interval: ${payload.calculatedTimeLagMonths} months nominal. Permutation Survival: ${(payload.permutationSurvivalRate * 100).toFixed(2)}%.`;

    // 2. التوجيه الديناميكي للبيانات بناءً على نوع التصنيف الوارد في الـ Payload

    const structuredSignal = {

      externalReference: payload.patternId,

      evidenceUrl: `system://inference-engine/patterns/${payload.patternId}`,

      evidenceClass: "structured_api" as const,

      

      // تصحيح هيكلي: إلغاء فرض كلمة "demand" وتمرير التصنيف عبر دالة توجيه مرنة

      signalType: this.routeSignalClassification(payload.signalClassification), 

      

      summary: compiledSummary,

      metrics: {

        impactDelta: payload.estimatedImpactMagnitude,

        horizonMonths: payload.calculatedTimeLagMonths,

        robustness: payload.statisticalRobustnessScore,

        // دمج المعرفات العامة ديناميكياً لتتوافق مع مفاتيح قاعدة البيانات الأساسية للمحرك الأب

        entityTargetKey: `${payload.targetEntity.domainSpace}_${payload.targetEntity.primaryIdentifier}`

      }

    };

    // 3. ضخ الإشارة الجاهزة إلى جداول الحوكمة والتشغيل للمحرك الحالي

    await this.submit(sourceId, structuredSignal);

  }

  private routeSignalClassification(classification: string): "supply" | "demand" {

    // أنبوب توجيه لضمان المواءمة مع قيود بيئة التشغيل الأساسية للنظام الأب

    if (classification === "supply") return "supply";

    if (classification === "demand") return "demand";

    

    // الصدمات النظامية ومعدلات الضغط يتم توجيهها افتراضياً كأحمال ضغط على الطلب

    return "demand";

  }

}

5.1 Technical Interface Architecture (Decoupled TypeScript Interface)
The updated CausalSignalAdapter decouples the core analytical output from location/item semantics and replaces hardcoded classifications with open-ended, dynamic parameters.

// src/connectors/CausalSignalAdapter.ts

import { DirectObservationAdapter } from "market-intelligence-engine";

export interface StructuralPayload {

  patternId: string;

  triggerEventClass: string;

  sourceContext: string;

  

  // Correction: Generic target descriptor decoupling the engine from item/location specifics

  targetEntity: {

    domainSpace: string;        // e.g., "geography_inventory", "financial_index", "asset_class"

    primaryIdentifier: string;  // e.g., "in-zone-4", "ticker-xyz", "commodity-faci"

    secondaryIdentifier?: string;

  };

  

  calculatedTimeLagMonths: number;

  

  // Correction: Separating mathematical significance from causality claims

  statisticalRobustnessScore: number;  // Out-of-sample persistence (0.00 - 1.00)

  permutationSurvivalRate: number;     // Pass rate against adversarial time shuffling

  

  estimatedImpactMagnitude: number;

  

  // Correction: Dynamic routing mapping rather than a hardcoded signal type

  signalClassification: "supply" | "demand" | "pressure_modifier" | "systemic_shock";

  timestampDiscovered: string;

}

export class CausalSignalAdapter extends DirectObservationAdapter {

  public readonly id = "structural_autonomous_inference_adapter";

  /**

   * Transforms an abstract statistical pattern into a dynamically routed

   * observation record within the core storage framework.

   */

  public async ingestAutonomousPattern(

    sourceId: string, 

    payload: StructuralPayload, 

    runtimeContext: any

  ): Promise<void> {

    

    // 1. Strict validation using the decoupled robustness thresholds

    if (payload.permutationSurvivalRate < 0.99 || payload.statisticalRobustnessScore < 0.99) {

      throw new Error(`Admission Denied: Signal payload failed to survive the adversarial shuffle verification threshold.`);

    }

    const compiledSummary = `[STRUCTURAL PATTERN ISOLATED] Precedence relationship verified from trigger [${payload.triggerEventClass}] to target space [${payload.targetEntity.domainSpace}::${payload.targetEntity.primaryIdentifier}]. Transmission interval: ${payload.calculatedTimeLagMonths} months nominal. Permutation Survival: ${(payload.permutationSurvivalRate * 100).toFixed(2)}%.`;

    // 2. Dynamic routing configuration based on the payload's classification

    const structuredSignal = {

      externalReference: payload.patternId,

      evidenceUrl: `system://inference-engine/patterns/${payload.patternId}`,

      evidenceClass: "structured_api" as const,

      

      // Correction: Removed hardcoded "demand" type allocation

      signalType: this.routeSignalClassification(payload.signalClassification), 

      

      summary: compiledSummary,

      metrics: {

        impactDelta: payload.estimatedImpactMagnitude,

        horizonMonths: payload.calculatedTimeLagMonths,

        robustness: payload.statisticalRobustnessScore,

        // Map generic entity identifiers natively into store parameters

        entityTargetKey: `${payload.targetEntity.domainSpace}_${payload.targetEntity.primaryIdentifier}`

      }

    };

    await this.submit(sourceId, structuredSignal);

  }

  private routeSignalClassification(classification: string): "supply" | "demand" {

    // Dynamic mapping pipeline ensuring compatibility with parent runtime constraints

    if (classification === "supply") return "supply";

    if (classification === "demand") return "demand";

    

    // Systemic shocks or pressure modifiers default to tracking systemic demand loads

    return "demand";

  }

}

ملاحظة تنفيذية نهائية — Non-Negotiable Execution Constraints
هذه الوثيقة هي المرجع التنفيذي المعتمد للإصدار v2.0. أثناء التنفيذ يجب الحفاظ على جميع المبادئ والحدود الواردة فيها، ولا يجوز تحويل أي قرار تنفيذي مؤقت إلى افتراض دلالي أو علمي غير مثبت.

البيانات الخام هي السجل المرجعي غير القابل للاستبدال. لا يجوز حذفها أو استبدالها بقيم مستنتجة أو مُعاد تشكيلها. أي تحويل أو تطبيع أو تجميع يجب أن ينتج طبقة تحليلية مشتقة مع الاحتفاظ بالمصدر الأصلي.

لا يجوز استخدام Forward-Fill أو Interpolation أو أي Imputation على نحو شامل. أي معالجة للفجوات يجب أن تكون خاصة بنوع المتغير، ومبررة رياضياً، ومسجلة صراحة باعتبارها قيمة مشتقة وليست مشاهدة أصلية. الأحداث المنفصلة لا يجوز تحويلها إلى حالات مستمرة بهذه الطرق.

الشهرية هي التواتر التحليلي الأساسي، وليست قيداً على دقة البيانات. تحتفظ البيانات اليومية/الساعية/دقيقة الحدث بدقتها الأصلية عندما تكون ضرورية لفهم الحدث أو تحديد التأخر الزمني. لا يجوز فقدان الدقة الأصلية بسبب عملية المحاذاة الشهرية.

يجب تسجيل التغطية على مستوى المتغير والسلسلة، وليس المصدر فقط. يجب حفظ start_date, end_date, frequency, missing_gaps, revision/version, وخصائص التغطية ذات الصلة لكل سلسلة مستخدمة.

الأحداث السياسية والجيوسياسية والعسكرية والبيئية والكوارث والقرارات والاغتيالات والعقوبات والإضرابات والانقلابات وغيرها ليست مجرد أرقام مشتقة. يجب الاحتفاظ بالحدث الأصلي وسياقه وزمنه ومصدره، ويمكن اشتقاق density/intensity/sentiment وغيرها كميزات إضافية فقط.

لا يجوز إدخال علاقات سببية أو DAGs مسبقة إلى محرك الاكتشاف. القيد المسموح مسبقاً هو اتجاه الزمن فقط. أي علاقة بين المتغيرات يجب أن تكون ناتجة عن عملية الاكتشاف والتحقق.

يجب الفصل دائماً بين الارتباط، الأسبقية الزمنية، تدفق المعلومات، المتانة الإحصائية، واكتشاف السببية. اجتياز permutation test أو out-of-sample validation لا يساوي إثبات سببية في العالم الحقيقي. يجب أن تعكس أسماء الحقول والتقارير هذا الفصل.

كل Pattern يجب أن يحمل provenance قابلاً لإعادة الإنتاج. يجب أن يكون بالإمكان معرفة البيانات والمدخلات والإصدارات والتحويلات والخوارزمية والـparameters وفترة التدريب/الاكتشاف وفترة الاختبار التي أدت إلى اكتشافه وقبوله.

لا يجوز استخدام البيانات المستقبلية في اكتشاف أو تقييم نمط تاريخي. يجب منع look-ahead bias وdata leakage، ويجب أن يتم أي backtesting وفق حدود زمنية تحترم ما كان متاحاً فعلياً عند النقطة التاريخية محل الاختبار.

المصادر المتعارضة لا تُسوّى قسراً. عندما تقدم مصادر موثوقة قياسات أو روايات مختلفة، تحفظ الاختلافات وتُنسب إلى مصادرها. لا يجوز اختيار قيمة واحدة لمجرد جعل البيانات متسقة دون قاعدة موثقة.

الـacademic datasets مثل FRED-MD/FRED-QD وMonash وM4/M5 هي بيئة اختبار ومعايرة، وليست بديلاً عن البيانات الواقعية. نجاح الخوارزمية عليها لا يُعتبر دليلاً على نجاحها في البيانات العالمية الحقيقية.

أي mapping مفروض بسبب توافق واجهة market-intelligence-engine يجب اعتباره Compatibility Mapping فقط. وبالأخص تحويل pressure_modifier أو systemic_shock إلى demand داخل الواجهة الحالية لا يعني أن معناها العلمي أو الدلالي هو demand. يجب عدم السماح لهذا القيد البرمجي بتغيير معنى البيانات أو الأنماط.

لا يتم إدخال Pattern إلى التشغيل الآلي لمجرد اكتشافه. يجب أن يمر أولاً عبر مراحل الاكتشاف، التحقق، الاختبار خارج العينة، مقاومة التشويش، وتوثيق provenance قبل أن يصبح إشارة تشغيلية.

لا يُسمح بإعادة تصميم المشروع أثناء التنفيذ لمجرد ظهور حل تقني أسهل. أي تغيير معماري جوهري يجب أن يستند إلى دليل تنفيذي أو اختبار يثبت عدم صلاحية التصميم الحالي، وليس إلى تفضيل شخصي أو تحسين شكلي.

الهدف النهائي ليس إنتاج Forecast جميل، بل اكتشاف أنماط تاريخية قابلة لإعادة الإنتاج ثم تحويل الأنماط التي تثبت متانتها إلى آليات مراقبة وتشغيل استباقي. أي نموذج Forecasting أو Causal Discovery أو Information Flow يُستخدم كأداة داخل هذه المنظومة، وليس كبديل عن المنظومة نفسها.
